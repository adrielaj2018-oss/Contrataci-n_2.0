<script>
async function activarCamara(){
 try{
   const stream=await navigator.mediaDevices.getUserMedia({video:true});
   document.getElementById('video').srcObject=stream;
   document.getElementById('estado').innerText='Cámara activada correctamente';
 }catch(e){
   document.getElementById('estado').innerText='Error activando cámara: '+e;
 }
}
function capturar(){
 const video=document.getElementById('video');
 const canvas=document.getElementById('canvas');
 canvas.width=video.videoWidth;
 canvas.height=video.videoHeight;
 canvas.getContext('2d').drawImage(video,0,0);
 const data=canvas.toDataURL('image/png');
 localStorage.setItem('evidencia_facial',data);
 document.getElementById('estado').innerText='Evidencia facial guardada';
}
